  TextInput, StyleSheet, SafeAreaView,

const initialData = {
  clients: [
    { id: 1, name: 'Jean M.', phone: '6XX XXX XXX' },
    { id: 2, name: 'Entreprise ABC', phone: '6XX XXX XXX' }
  ],
  projects: [
    { id: 1, name: 'Construction Villa', amount: 2500000 },
    { id: 2, name: 'Rénovation maison', amount: 850000 }
  ],
  quotes: [],
  invoices: [],
  materials: [
    { id: 1, name: 'Ciment', amount: 5000 },
    { id: 2, name: 'Fer à béton', amount: 4500 }
  ]
};

const sections = [
  { key: 'dashboard', icon: '⌂' },
  { key: 'clients', icon: '♙' },
  { key: 'projects', icon: '▦' },
  { key: 'quotes', icon: '▤' },
  { key: 'invoices', icon: '▣' },
  { key: 'materials', icon: '◈' }
];

export default function App() {
  const [lang, setLang] = useState('fr');
  const [page, setPage] = useState('dashboard');
  const [data, setData] = useState(initialData);
  const [modal, setModal] = useState(false);
  const [form, setForm] = useState({
    name: '', phone: '', amount: ''
  });
  const [search, setSearch] = useState('');

  const t = translations[lang];

  const labels = {
    clients: t.clients,
    projects: t.projects,
    quotes: t.quotes,
    invoices: t.invoices,
    materials: t.materials
  };

  const formatMoney = (value) =>
    Number(value || 0).toLocaleString('fr-FR') + ' FCFA';

  const totalRevenue = data.invoices.reduce(
    (sum, item) => sum + Number(item.amount || 0), 0
  );

  const openForm = () => {
    setForm({ name: '', phone: '', amount: '' });
    setModal(true);
  };

  const saveItem = () => {
    if (!form.name.trim()) {
      Alert.alert('MAG BTP', t.name + ' requis');
      return;
    }

    const item = {
      id: Date.now(),
      name: form.name.trim(),
      phone: form.phone.trim(),
      amount: Number(form.amount || 0)
    };

    setData(old => ({
      ...old,
      [page]: [...old[page], item]
    }));

    setModal(false);
    Alert.alert('MAG BTP', t.save + ' ✓');
  };

  const pageTitle = page === 'dashboard'
    ? t.dashboard : labels[page] || t.settings;

  const renderDashboard = () => (
    <>
      <View style={styles.welcome}>
        <Text style={styles.welcomeSmall}>MAG BTP ⚡</Text>
        <Text style={styles.welcomeTitle}>{t.welcome}</Text>
        <Text style={styles.welcomeSub}>{t.overview}</Text>
      </View>

      <Text style={styles.sectionTitle}>{t.dashboard}</Text>

      <View style={styles.stats}>
        <Stat label={t.revenue}
          value={formatMoney(totalRevenue)}
          icon="↗" />
        <Stat label={t.active}
          value={String(data.projects.length)}
          icon="⌂" />
        <Stat label={t.customers}
          value={String(data.clients.length)}
          icon="♙" />
        <Stat label={t.pending}
          value={String(data.quotes.length)}
          icon="▤" />
      </View>

      <Text style={styles.sectionTitle}>{t.quick}</Text>
      <View style={styles.quickGrid}>
        {['clients', 'projects', 'quotes', 'invoices'].map(key => (
          <TouchableOpacity
            key={key}
            style={styles.quickButton}
            onPress={() => {
              setPage(key);
              setForm({ name: '', phone: '', amount: '' });
              setModal(true);
            }}>
            <Text style={styles.quickIcon}>＋</Text>
            <Text style={styles.quickText}>
              {t['add' + key.charAt(0).toUpperCase() + key.slice(1)] ||
               labels[key]}
            </Text>
          </TouchableOpacity>
        ))}
      </View>

      <Text style={styles.sectionTitle}>{t.recent}</Text>
      <View style={styles.card}>
        {data.projects.slice(-3).reverse().map(item => (
          <View key={item.id} style={styles.activity}>
            <View style={styles.activityIcon}>
              <Text>⌂</Text>
            </View>
            <View style={{ flex: 1 }}>
              <Text style={styles.itemName}>{item.name}</Text>
              <Text style={styles.muted}>{t.projects}</Text>
            </View>
            <Text style={styles.orange}>En cours</Text>
          </View>
        ))}
        {data.projects.length === 0 &&
          <Text style={styles.muted}>{t.noData}</Text>}
      </View>
    </>
  );

  const renderList = () => {
    const items = data[page] || [];
    const filtered = items.filter(item =>
      item.name.toLowerCase().includes(search.toLowerCase())
    );

    return (
      <>
        <TextInput
          style={styles.search}
          placeholder={t.search}
          value={search}
          onChangeText={setSearch}
        />
        <TouchableOpacity style={styles.addButton} onPress={openForm}>
          <Text style={styles.addText}>＋  {labels[page]}</Text>
        </TouchableOpacity>

        {filtered.length === 0 ?
          <View style={styles.card}>
            <Text style={styles.muted}>{t.empty}</Text>
          </View>
          : filtered.map(item => (
            <View style={styles.listCard} key={item.id}>
              <View style={styles.activityIcon}>
                <Text style={{ color: ORANGE }}>◆</Text>
              </View>
              <View style={{ flex: 1 }}>
                <Text style={styles.itemName}>{item.name}</Text>
                {item.phone ? (
                  <Text style={styles.muted}>{item.phone}</Text>
                ) : null}
                {item.amount > 0 ? (
                  <Text style={styles.orange}>
                    {formatMoney(item.amount)}
                  </Text>
                ) : null}
              </View>
            </View>
          ))}
      </>
    );
  };

  return (
    <SafeAreaView style={styles.safe}>
      <StatusBar barStyle="light-content" backgroundColor={NAVY} />

      <View style={styles.header}>
        <View>
          <Text style={styles.logo}>MAG <Text style={{ color: ORANGE }}>BTP.</Text></Text>
          <Text style={styles.headerSub}>BUILD • MANAGE • SUCCEED</Text>
        </View>
        <TouchableOpacity
          style={styles.langButton}
          onPress={() => setLang(lang === 'fr' ? 'en' : 'fr')}>
          <Text style={styles.langText}>
            {lang === 'fr' ? '🇫🇷 FR' : '🇬🇧 EN'}
          </Text>
        </TouchableOpacity>
      </View>

      <View style={styles.pageHeader}>
        <Text style={styles.pageTitle}>{pageTitle}</Text>
        <Text style={styles.pageDate}>MAG BTP • 2026</Text>
      </View>

      <ScrollView
        style={styles.content}
        contentContainerStyle={{ paddingBottom: 30 }}
        keyboardShouldPersistTaps="handled">
        {page === 'dashboard' ? renderDashboard() : renderList()}
      </ScrollView>

      <View style={styles.bottomNav}>
        {sections.slice(0, 5).map(item => (
          <TouchableOpacity
            key={item.key}
            style={styles.navItem}
            onPress={() => {
              setPage(item.key);
              setSearch('');
            }}>
            <Text style={[
              styles.navIcon,
              page === item.key && styles.navActive
            ]}>{item.icon}</Text>
            <Text style={[
              styles.navLabel,
              page === item.key && styles.navActive
            ]}>{t[item.key]}</Text>
          </TouchableOpacity>
        ))}
      </View>

      {modal && (
        <View style={styles.modalOverlay}>
          <View style={styles.modal}>
            <Text style={styles.modalTitle}>
              {labels[page] || t.save}
            </Text>
            <TextInput
              style={styles.input}
              placeholder={t.name + ' / ' + t.title}
              value={form.name}
              onChangeText={v => setForm({ ...form, name: v })}
            />
            {page === 'clients' && (
              <TextInput
                style={styles.input}
                placeholder={t.phone}
                keyboardType="phone-pad"
                value={form.phone}
                onChangeText={v => setForm({ ...form, phone: v })}
              />
            )}
            {['projects', 'quotes', 'invoices', 'materials'].includes(page) && (
              <TextInput
                style={styles.input}
                placeholder={t.amount}
                keyboardType="numeric"
                value={form.amount}
                onChangeText={v => setForm({ ...form, amount: v })}
              />
            )}
            <TouchableOpacity style={styles.addButton} onPress={saveItem}>
              <Text style={styles.addText}>{t.save}</Text>
            </TouchableOpacity>
            <TouchableOpacity onPress={() => setModal(false)}>
              <Text style={styles.cancel}>{t.cancel}</Text>
            </TouchableOpacity>
          </View>
        </View>
      )}
    </SafeAreaView>
  );
}

function Stat({ label, value, icon }) {
  return (
    <View style={styles.statCard}>
      <Text style={styles.statIcon}>{icon}</Text>
      <Text style={styles.statValue}>{value}</Text>
      <Text style={styles.statLabel}>{label}</Text>
    </View>
  );
}
